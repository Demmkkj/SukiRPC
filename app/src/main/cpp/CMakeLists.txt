cmake_minimum_required(VERSION 3.22.1)
project(nuvio_rpc LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(discord_partner_sdk REQUIRED CONFIG)
add_library(nuvio_rpc SHARED rpc.cpp)
target_link_libraries(nuvio_rpc discord_partner_sdk::discord_partner_sdk log)

# Only the private quality APK variants select these flags.
if(SUKI_SANITIZER STREQUAL "hwasan")
    target_compile_options(nuvio_rpc PRIVATE -fsanitize=hwaddress -fno-omit-frame-pointer)
    target_link_options(nuvio_rpc PRIVATE -fsanitize=hwaddress)
elseif(SUKI_SANITIZER STREQUAL "ubsan")
    target_compile_options(nuvio_rpc PRIVATE -fsanitize=undefined -fno-sanitize-recover=undefined)
    target_link_options(nuvio_rpc PRIVATE -fsanitize=undefined)
elseif(DEFINED SUKI_SANITIZER)
    message(FATAL_ERROR "Unknown SukiRPC sanitizer variant")
endif()
